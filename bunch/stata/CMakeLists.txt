cmake_minimum_required(VERSION 3.20)
include("${CMAKE_CURRENT_LIST_DIR}/../../common/macos_deployment.cmake")
project(stata_plugin LANGUAGES C CXX)
add_subdirectory(../core core)
# Extend the core target in this wrapper project; no core files are changed.
target_sources(specbunch PRIVATE plugin.cpp entry.c)
include("${CMAKE_CURRENT_LIST_DIR}/../../common/third_party/stata_spi.cmake")
sbc_stata_spi(specbunch)
set_target_properties(specbunch PROPERTIES PREFIX "" SUFFIX ".plugin" OUTPUT_NAME "specbunch_${STATA_PLATFORM}")
if(APPLE)
 target_compile_definitions(specbunch PRIVATE SYSTEM=APPLEMAC)
 set_property(TARGET specbunch PROPERTY LINK_OPTIONS "LINKER:-exported_symbol,_pginit" "LINKER:-exported_symbol,_stata_call" "LINKER:-dead_strip")
elseif(UNIX)
 target_compile_definitions(specbunch PRIVATE SYSTEM=OPUNIX)
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/plugin.map" "{ global: pginit; stata_call; local: *; };\n")
 set_property(TARGET specbunch PROPERTY LINK_OPTIONS "LINKER:--version-script=${CMAKE_CURRENT_BINARY_DIR}/plugin.map" "LINKER:--gc-sections")
endif()
# entry.c explicitly exports the SPI entry points despite core hidden defaults.
add_custom_target(stata_plugin DEPENDS specbunch)
